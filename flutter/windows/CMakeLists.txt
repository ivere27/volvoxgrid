cmake_minimum_required(VERSION 3.14)
project(volvoxgrid LANGUAGES CXX)

set(VOLVOXGRID_PLATFORM windows)
set(VOLVOXGRID_NATIVE_NAME volvoxgrid.dll)
if(CMAKE_GENERATOR_PLATFORM MATCHES "^(ARM64|arm64)$" OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|aarch64)$")
  message(FATAL_ERROR "VolvoxGrid: Maven Windows ARM64 binaries are not available. Use a supported x64 target.")
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(VOLVOXGRID_NATIVE_ARCH windows-x86_64)
else()
  set(VOLVOXGRID_NATIVE_ARCH windows-x86)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../native/resolve.cmake")
set(volvoxgrid_bundled_libraries "${VOLVOXGRID_RESOLVED_LIBRARY}" PARENT_SCOPE)
